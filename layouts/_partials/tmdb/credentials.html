{{- /* 配置优先；配置均为空时，读取配置指定的凭据文件（默认 .tmdb）。 */ -}} {{- $token := .token | default "" -}} {{- $key := .api_key | default "" -}} {{- if ne (printf "%T" $token) "string" -}}{{- errorf "tmdb: api.token must be a string" -}}{{- $token = "" -}}{{- end -}} {{- if ne (printf "%T" $key) "string" -}}{{- errorf "tmdb: api.api_key must be a string" -}}{{- $key = "" -}}{{- end -}} {{- $token = strings.TrimSpace $token -}}{{- $key = strings.TrimSpace $key -}} {{- if and (not $token) (not $key) -}} {{- $file := .credentials_file | default ".tmdb" -}} {{- $valid := eq (printf "%T" $file) "string" -}} {{- if $valid -}} {{- $file = path.Clean $file -}} {{- $valid = not (or (strings.HasPrefix $file "/") (strings.HasPrefix $file "../") (eq $file "..") (strings.Contains $file ":") (strings.Contains $file `\`)) -}} {{- end -}} {{- if not $valid -}} {{- errorf "tmdb: api.credentials_file must be a file path relative to the site root" -}} {{- else if os.FileExists $file -}} {{- $parsed := try (transform.Unmarshal (dict "format" "toml") (os.ReadFile $file)) -}} {{- if or $parsed.Err (not (reflect.IsMap $parsed.Value)) -}} {{- /* 不打印解析错误，避免将含凭据的原始行写入日志。 */ -}} {{- errorf "tmdb: .tmdb must be valid TOML with string token and api-key fields" -}} {{- else -}} {{- $values := $parsed.Value -}} {{- $token = index $values "token" | default "" -}} {{- $key = index $values "api-key" | default "" -}} {{- if ne (printf "%T" $token) "string" -}}{{- errorf "tmdb: .tmdb token must be a string" -}}{{- $token = "" -}}{{- end -}} {{- if ne (printf "%T" $key) "string" -}}{{- errorf "tmdb: .tmdb api-key must be a string" -}}{{- $key = "" -}}{{- end -}} {{- $token = strings.TrimSpace $token -}}{{- $key = strings.TrimSpace $key -}} {{- end -}} {{- end -}} {{- end -}} {{- if $token -}}{{- $key = "" -}}{{- end -}} {{- return (dict "token" $token "key" $key) -}}
